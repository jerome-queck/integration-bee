\section{Learning goals}
\begin{enumerate}
  \item Use $n!=\int_{0}^{\infty}x^{n}e^{-x}\,dx$ on easy integer $n$.
  \item Convert $\int_{0}^{1}(\ln x)^{n}\,dx=(-1)^{n}n!$ by $x=e^{-t}$.
  \item Recognise Gamma-shaped integrals after a short $u$-sub.
\end{enumerate}

\section{Worked example --- \texttt{U14-D01} (\texttt{yang:2.4})}
\[
\int_{0}^{\infty}x^{5}e^{-x}\,dx.
\]
By the factorial formula: $5!=120$.

\section{Faded example}
Try U14-D03: $\int_{0}^{1}(\ln x)^{3}\,dx=(-1)^{3}3!=-6$.
