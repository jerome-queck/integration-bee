\section{Learning goals}
\begin{enumerate}
  \item Solve for $x$ in terms of $u$ when $du$ is missing (e.g.\ $u=\sqrt{x}\Rightarrow x=u^{2}$, $dx=2u\,du$).
  \item Integrate $e^{\sqrt{x}}$, $\sin(\ln x)$, $\cos(\sqrt{x})$ after a forced sub.
  \item Evaluate easy definite forced-$u$ integrals on $[0,1]$ or $[0,4]$.
\end{enumerate}

\section{Pedagogy}
Worked $\to$ faded $\to$ interleaved easy drills. Chat Flow 1.

\section{Worked example --- \texttt{U13-D01} (\texttt{yang:2.3})}
\[
\int e^{\sqrt{x}}\,dx.
\]
Let $u=\sqrt{x}$, so $x=u^{2}$, $dx=2u\,du$: $\int 2u e^{u}\,du=2e^{u}(u-1)+C=2e^{\sqrt{x}}(\sqrt{x}-1)+C$.

\section{Faded example}
Try U13-D04: $\int\cos\sqrt{x}\,dx$ with the same $u=\sqrt{x}$ forced pattern.
