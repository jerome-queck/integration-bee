\section{Learning goals}

\end{enumerate}

\section{Pedagogy}

\section{Worked --- \texttt{U06-D01} (\texttt{yang:1.11})}
\[
\int_0^1 x^3 e^{x^4}\,dx,\quad u=x^4\Rightarrow \frac14\int_0^1 e^u\,du=\frac{e-1}{4}.
\]

\section{Faded --- \texttt{U06-D03}}
\[
\int_e^{e^2}\frac{1}{x\ln x}\,dx.
\]
Scaffold: $u=\ln x$, $du=dx/x$.
