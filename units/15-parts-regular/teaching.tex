\section{Learning goals}
\begin{enumerate}
  \item Choose $u$/$dv$ on slightly trickier regular-round parts.
  \item Handle $\int x\ln(x+1)$, $\int e^{x}\sin x$, and $\int(\ln x)^{2}/x^{3}$.
  \item Evaluate one-step definite parts cleanly.
\end{enumerate}

\section{Worked example --- \texttt{U15-D01} (\texttt{yang:2.5})}
\[
\int_{0}^{1}x\ln(x+1)\,dx.
\]
Parts with $u=\ln(x+1)$, $dv=x\,dx$ (or expand series). Value $1/4$.
