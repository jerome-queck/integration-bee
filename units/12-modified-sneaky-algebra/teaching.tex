\section{Learning goals}
\begin{enumerate}
  \item Take a textbook skeleton and modify it (square a denominator; hide a $u$-sub inside parts).
  \item Factor cleverly (difference of squares under a radical denominator).
  \item Absorb polynomial factors into a single $u$-power.
  \item Center a product integral with a shift $u=x-a$.
\end{enumerate}

\section{Worked example A --- \texttt{U12-D01} (\texttt{yang:2.1})}
\[
\int e^{x}\,e^{e^{x}}\,dx.
\]
Let $u=e^{x}$, $du=e^{x}\,dx$: $\int e^{u}\,du=e^{e^{x}}+C$.

\section{Worked example B --- \texttt{U12-D05} (\texttt{yang:2.2})}
\[
\int_0^1 \frac{x-4}{\sqrt{x}-2}\,dx.
\]
Factor $x-4=(\sqrt{x}-2)(\sqrt{x}+2)$, so the integrand is $\sqrt{x}+2$ on $[0,1]$:
\[
\int_0^1(\sqrt{x}+2)\,dx=\Bigl[\tfrac23 x^{3/2}+2x\Bigr]_0^1=\tfrac83.
\]
