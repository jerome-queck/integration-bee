\section{Worked arcsin --- \texttt{U04-D01} (\texttt{yang:1.7})}
\[
\int_0^1\frac{1}{\sqrt{1-x^2}}\,dx = \arcsin x\Big|_0^1=\frac\pi2.
\]

\section{Worked quarter circle --- \texttt{U04-D02} (\texttt{yang:1.9})}
\[
\int_0^1\sqrt{1-x^2}\,dx = \frac\pi4
\]
(area of quarter unit disk; trig-sub $x=\sin\theta$ also works).

\section{Faded --- \texttt{U04-D03} (\texttt{yang:1.7})}
\[
\int_0^2\frac{1}{\sqrt{4-x^2}}\,dx.
\]
Scaffold: $x=2\sin\theta$ $\Rightarrow$ $\arcsin(x/2)$ from $0\to2$ gives $\pi/2$.
