\section{Worked --- \texttt{U05-D01} (\texttt{yang:1.8})}
\[
\frac{1}{(x+1)(x+2)}=\frac{1}{x+1}-\frac{1}{x+2},\quad
\int_0^1=\ln\frac43.
\]

\section{Faded --- \texttt{U05-D02}}
\[
\int_1^2\frac{1}{x(x+1)}\,dx.
\]
Scaffold: $\frac{1}{x}-\frac{1}{x+1}$. Finish to $\ln\frac43$.
